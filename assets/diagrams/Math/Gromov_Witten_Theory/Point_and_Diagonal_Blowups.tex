\begin{tikzpicture}[line cap=round, line join=round]
  %% the diagonal, drawn as a slightly tilted line
  \draw[line width=1pt] (-3,-.4)--(3,.4) node[right] {$\Delta$};
  %% the slice P^2 x {q} through (q,q); in-plane coordinates (y_1-x_1, y_2-x_2)
  \begin{scope}[x={(-.6cm,-.45cm)}, y={(0cm,1cm)}]
    \draw[black!50] (-2.9,-1.6)--(2.1,-1.6)--(2.1,2.4)--(-2.9,2.4)--cycle;
    \node[black!50, above right] at (-2.9,2.4) {$\mathbb{P}^2\times\{q\}$};
    \draw[black!50] (-1.5,0)--(1.5,0);
    \draw[black!50] (0,-1.5)--(0,1.1);
    %% auxiliary lines reading off the two components of the vector
    \draw[dashed, black!50] (-1.1,1)--(-1.1,0) node[midway, right=1pt, black] {$q_2-p_2$};
    \draw[dashed, black!50] (-1.1,1)--(0,1) node[midway, above left=1pt, black] {$q_1-p_1$};
    %% the vector from (p,q) into the diagonal
    \draw[->, accent1, line width=1pt, shorten >=2.5pt] (-1.1,1)--(0,0);
    \fill (-1.1,1) circle[radius=1.6pt] node[above right=1pt] {$(p,q)$};
  \end{scope}
  \fill[accent1] (0,0) circle[radius=2pt] node[below right=1pt] {$(q,q)$};
\end{tikzpicture}

%%FIG the line through q and p tends to the tangent line as q approaches p (Point_and_Diagonal_Blowups-2.svg)
\begin{tikzpicture}[line cap=round, line join=round]
  %% secant lines through p and q, q', and the limiting tangent line
  \foreach \s in {2,1.1}{
    \pgfmathsetmacro{\m}{.35*\s-.08*\s*\s}
    \draw[accent1!55, line width=.8pt] (-2.2,{-2.2*\m})--(2.7,{2.7*\m});
  }
  \draw[accent1, line width=1pt] (-2.2,0)--(2.7,0);
  \node[accent1, right] at (2.7,0) {$\ell$};
  %% a curve through p, parametrized as (t, .35 t^2 - .08 t^3)
  \draw[black!50, line width=.9pt] plot[domain=-1.9:2.4, variable=\t, samples=120]
    ({\t},{.35*\t*\t-.08*\t*\t*\t});
  \foreach \s/\lab in {2/{q},1.1/{q'}}{
    \pgfmathsetmacro{\m}{.35*\s-.08*\s*\s}
    \fill ({\s},{\s*\m}) circle[radius=1.6pt];
    \node[above left=1pt] at ({\s},{\s*\m}) {$\lab$};
  }
  \fill (0,0) circle[radius=2pt];
  \node[below=2pt] at (0,0) {$p$};
\end{tikzpicture}

%%FIG blowing up a point separates the lines through it (Point_and_Diagonal_Blowups-4.svg)
\begin{tikzpicture}[line cap=round]
  %% the plane: four lines through q
  \foreach \ang in {15,60,105,150}{
    \draw[line width=.8pt] ({-1.6*cos(\ang)},{-1.6*sin(\ang)})--({1.6*cos(\ang)},{1.6*sin(\ang)});
  }
  \fill[accent1] (0,0) circle[radius=2pt];
  \node[below right=1pt] at (0,0) {$q$};
  \node at (0,-2.15) {$\mathbb{P}^2$};

  %% the blowup: E horizontal, strict transforms are the vertical fibers
  \begin{scope}[shift={(6,0)}]
    \draw[accent1, line width=1pt] (-2,0)--(2,0);
    \node[accent1, right] at (2,0) {$E$};
    \foreach \x in {-1.35,-.45,.45,1.35}{
      \draw[line width=.8pt] (\x,-1.4)--(\x,1.4);
      \fill (\x,0) circle[radius=1.3pt];
    }
    \node at (0,-2.15) {$\mathrm{Bl}_q\mathbb{P}^2$};
  \end{scope}

  \draw[->, black!50, line width=.8pt] (3.4,0)--(2.1,0) node[midway, above] {$\sigma$};
\end{tikzpicture}

%%FIG strict transforms of a nodal and a cuspidal cubic (Point_and_Diagonal_Blowups-5.svg)
\begin{tikzpicture}[line cap=round, line join=round]
  %% downstairs: the nodal cubic y^2 = x^2(x+1) and the cuspidal cubic y^2 = x^3
  \draw[line width=.9pt] plot[domain=-1.55:1.55, variable=\t, samples=120]
    ({\t*\t-1},{.75*\t*(\t*\t-1)});
  \fill[accent1] (0,0) circle[radius=2pt];
  \node[below left=1pt] at (0,0) {$q$};

  \begin{scope}[shift={(5.2,0)}]
    \draw[line width=.9pt] plot[domain=-1.35:1.35, variable=\t, samples=120]
      ({\t*\t},{.75*\t*\t*\t});
    \fill[accent1] (0,0) circle[radius=2pt];
    \node[left=2pt] at (0,0) {$q$};
  \end{scope}

  %% upstairs, in the chart (x,v) with y = xv: E = {x = 0} is vertical
  \begin{scope}[shift={(0,4.6)}]
    \draw[accent1, line width=1pt] (0,-1.75)--(0,1.75);
    \node[accent1, above] at (0,1.75) {$E$};
    \draw[line width=.9pt] plot[domain=-1.55:1.55, variable=\t, samples=120]
      ({\t*\t-1},{\t});
    \fill (0,1) circle[radius=1.6pt];
    \fill (0,-1) circle[radius=1.6pt];
  \end{scope}

  \begin{scope}[shift={(5.2,4.6)}]
    \draw[accent1, line width=1pt] (0,-1.75)--(0,1.75);
    \node[accent1, above] at (0,1.75) {$E$};
    \draw[line width=.9pt] plot[domain=-1.35:1.35, variable=\t, samples=120]
      ({\t*\t},{\t});
    \fill (0,0) circle[radius=1.6pt];
  \end{scope}

  \draw[->, black!50, line width=.8pt] (.2,2.55)--(.2,1.65);
  \draw[->, black!50, line width=.8pt] (5.4,2.55)--(5.4,1.65);
\end{tikzpicture}
